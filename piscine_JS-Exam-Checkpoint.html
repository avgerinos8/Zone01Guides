perfect-num

Instructions

Write a function that takes a number (n) and returns:

  • true → if the number is a perfect number.
  • false → if it’s not a perfect number.

What’s a perfect number?

A perfect number is a positive number where the sum of all its proper divisors (numbers that divide it except itself) equals the number.

For example:

  • 6 → divisors: 1, 2, 3 → 1 + 2 + 3 = 6 → perfect!
  • 28 → divisors: 1, 2, 4, 7, 14 → 1 + 2 + 4 + 7 + 14 = 28 → perfect!
  • 12 → divisors: 1, 2, 3, 4, 6 → 1 + 2 + 3 + 4 + 6 = 16 → not perfect.

Expected function

function isPerfectNum(n) {

}

Usage

Here is a possible program to test your function:

console.log(isPerfectNum(6))
console.log(isPerfectNum(28))
console.log(isPerfectNum(12))

And its output:

$ node main.js
true
true
false
$

Tip:

To check if a number is perfect:

  • Start with the total as 0.
  • For every number less than n (from 1 to n - 1): If it divides n with no remainder, add it to the total.
  • At the end, if the total equals n, it’s a perfect number.
function isPerfectNum(n) {
if (n <= 1) return false
let divisors = [1]
for (let i = 2; i < n; i++) {
if (n % i === 0) {
divisors.push(i)
}
}
let summary = divisors.reduce(summ, 0)
let verdict
summary === n ? verdict = true : verdict = false
return verdict
}
function summ(acc, number) {
return acc + number
}

sentence-pyramid

Instructions

Write a function that takes a string (sentence) and prints a pyramid where:

  • The first line prints the first word.
  • The second line prints the first two words.
  • The third line prints the first three words.
  • And so on… until the full sentence is printed.

Expected function

function sentencePyramid(sentence) {

}

Usage

Here is a possible program to test your function:

sentencePyramid("This is a simple sentence pyramid")

And its output:

$ node main.js
This
This is
This is a
This is a simple
This is a simple sentence
This is a simple sentence pyramid
$

Tip:

To build the pyramid:

  • Split the sentence into words → use .split(" ").
  • Start from the first word and print lines adding one more word each time.
  • Stop when all words are printed.
function sentencePyramid(str) {
if (!str) {
return
}
let words = str.trim().split(/\s+/)
for (let i = 1; i <= words.length; i++) {
console.log(words.slice(0, i).join(" "))
}
}

bubble-sort-analyzer

Instructions

Write a function that takes:

  • An array (arr)
  • A comparator function (comparator) to compare two numbers.

Your function should:

  1. Sort the array using the bubble sort algorithm.
  2. Count:
    • iterations → total number of comparisons/steps.
    • swaps → total times two numbers were switched.
  3. Return an object with:
    • The sorted array.
    • The number of iterations.
    • The number of swaps.

Expected function

function bubbleSortAnalyzer(arr, comparator) {

}

Usage

Here is a possible program to test your function:

const comparator = (a, b) => a - b

const result = bubbleSortAnalyzer([4, 2, 7, 1, 3], comparator)

console.log(result)

And its output:

$ node main.js
{
  sortedArray: [ 1, 2, 3, 4, 7 ],
  iterations: 10,
  swaps: 6
}
$

Tip — How Bubble Sort Works:

  • Compare each pair of items in the array.
  • If they’re in the wrong order (according to the comparator), swap them.
  • Repeat this process for the whole array multiple times until no swaps are needed.
  • On every pass, the largest item "bubbles up" to the end.

Tip — About the Comparator:

The comparator decides how to compare two numbers:

(a, b) => a - b
  • If result < 0 → a comes before b.
  • If result === 0 → a and b are equal.
  • If result > 0 → a comes after b.
function bubbleSortAnalyzer(arr, comparator) {
let a = [...arr]
let iterations = 0
let swaps = 0
let n = a.length
for (let i = 0; i < n - 1; i++) {
let swapped = false
for (let j = 0; j < n - 1 - i; j++) {
iterations++
if (comparator(a[j], a[j + 1]) > 0) {
let temp = a[j]
a[j] = a[j + 1]
a[j + 1] = temp
swaps++
swapped = true
}
}
if (!swapped) break
}
return {
sortedArray: a,
iterations,
swaps
}
}

flat-object

Instructions

Write a function flattenAndMap(obj, mapper) that:

  • Takes a deeply nested object obj where values might be objects or primitives.
  • Flattens this object so that each key in the resulting object represents the path to the value in the original object, using dot notation for nesting (e.g. user.address.city).
  • Applies a mapper function to each value (primitive) before placing it into the flattened result.
  • Returns an object with:
    • flattened: The flattened object after mapping values.
    • originalKeysCount: The total number of keys (including nested) found in the original object.
    • transformedKeysCount: The total number of keys whose values were transformed by the mapper.

Expected function

function flattenAndMap(obj, mapper) {

}

Usage

Here is a possible program to test your function:

const nestedObj = {
  user: {
    name: "Alice",
    address: {
      city: "Wonderland",
      zip: 12345,
    },
  },
  meta: {
    roles: ["admin", "editor"],
    active: true,
  },
}

const toUpperCaseStrings = (value) =>
  typeof value === "string" ? value.toUpperCase() : value

const result = flattenAndMap(nestedObj, toUpperCaseStrings)

console.log(result)

And its output:

$ node main.js
{
  flattened: {
    'user.name': 'ALICE',
    'user.address.city': 'WONDERLAND',
    'user.address.zip': 12345,
    'meta.roles': [ 'ADMIN', 'EDITOR' ],
    'meta.active': true
  },
  originalKeysCount: 5,
  transformedKeysCount: 5
}
$

Notes:

  • The mapper function should be applied to every value, regardless of type. If it's not a string or the mapper doesn't change it, that's fine.
  • Arrays are considered values too, and should be mapped. If mapping arrays, apply the mapper to each element.
  • Count all final flattened paths as keys in the resulting object.
  • originalKeysCount is how many ultimate primitive or array-containing keys were found in the original nested structure.
  • transformedKeysCount is how many values ended up being transformed by the mapper (even if that transformation doesn't change the value).
function flattenAndMap(obj, mapper) {
let flattened = {}
let originalKeysCount = 0
let transformedKeysCount = 0
function traverse(current, path) {
for (let key in current) {
if (!Object.prototype.hasOwnProperty.call(current, key)) continue
let val = current[key]
let newPath = path ? path + "." + key : key
if (val !== null && typeof val === "object" && !Array.isArray(val)) {
traverse(val, newPath)
} else {
originalKeysCount++
if (Array.isArray(val)) {
flattened[newPath] = val.map(item => mapper(item))
} else {
flattened[newPath] = mapper(val)
}
transformedKeysCount++
}
}
}
traverse(obj, "")
return { flattened, originalKeysCount, transformedKeysCount }
}
